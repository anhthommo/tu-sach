Bước 1: Mở file HTML, tìm phần CONFIG ở đầu JS và điền 3 giá trị:
const SUPABASE_URL  = 'https://xxx.supabase.co';  // URL project của bạn
const SUPABASE_ANON = 'eyJhbGci...';               // Anon/public key
const ADMIN_EMAIL   = 'your@email.com';            // Email của bạn (admin)
Bước 2: Trong Supabase SQL Editor, chạy SQL này để tạo bảng và RLS:
-- 1. Tạo bảng books với user_id
CREATE TABLE books (
  id          BIGSERIAL PRIMARY KEY,
  user_id     UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  title       TEXT NOT NULL,
  author      TEXT NOT NULL,
  year        INTEGER,
  genre       TEXT,
  status      TEXT DEFAULT 'unread',
  rating      INTEGER DEFAULT 0,
  cover       TEXT,
  color       TEXT DEFAULT '#1a1008',
  text_color  TEXT DEFAULT '#e8d5a0',
  note        TEXT,
  quotes      JSONB DEFAULT '[]',
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Bật Row Level Security
ALTER TABLE books ENABLE ROW LEVEL SECURITY;

-- 3. Mọi người đều đọc được (để hiển thị tủ admin công khai)
CREATE POLICY "public_read" ON books
  FOR SELECT USING (true);

-- 4. Chỉ owner mới insert/update/delete sách của mình
CREATE POLICY "owner_write" ON books
  FOR INSERT WITH CHECK (auth.uid() = user_id);
CREATE POLICY "owner_update" ON books
  FOR UPDATE USING (auth.uid() = user_id);
CREATE POLICY "owner_delete" ON books
  FOR DELETE USING (auth.uid() = user_id);

-- 5. RPC để app lấy admin_uid mà không cần service key
--    Thay 'your@email.com' bằng email admin của bạn
CREATE OR REPLACE FUNCTION get_admin_uid()
RETURNS UUID LANGUAGE sql SECURITY DEFINER AS $$
  SELECT id FROM auth.users
  WHERE email = 'your@email.com'
  LIMIT 1;
$$;
Bước 3: Đăng nhập bằng ADMIN_EMAIL → hệ thống tự lưu admin_uid vào localStorage → các user khác vào trang sẽ xem được tủ sách của bạn.
